        function copyCurrentCodeTab() {
            const text = repoFiles[activeCodeTab];
            const temp = document.createElement('textarea');
            temp.value = text;
            document.body.appendChild(temp);
            temp.select();
            document.execCommand('copy');
            document.body.removeChild(temp);
